const TEXT_MODEL = 'gemini-2.5-flash'; const IMAGE_MODEL = 'gemini-2.5-flash-image'; // AI Studio-la current image model name check pannikko function getKey_() { return PropertiesService.getScriptProperties().getProperty('GEMINI_API_KEY'); } function callGemini_(model, payload) { const url = 'https://generativelanguage.googleapis.com/v1beta/models/' + model + ':generateContent?key=' + getKey_(); const res = UrlFetchApp.fetch(url, { method: 'post', contentType: 'application/json', payload: JSON.stringify(payload), muteHttpExceptions: true }); return JSON.parse(res.getContentText()); } // ---- Story: returns 4 panels + a character/style description for image consistency ---- function generateStory_(idea) { const prompt = 'Create a 4-panel comic story from this idea: "' + idea + '".\n' + 'Return ONLY JSON in this exact shape:\n' + '{"characters":"short visual description of main characters (age, hair, clothes, colors)",' + '"panels":[{"title":"The Beginning","text":"1-2 sentences","speech":"one short dialogue line","scene":"visual description of what is drawn in this panel"},' + '{"title":"The Problem",...},{"title":"The Battle",...},{"title":"The Ending",...}]}'; const data = callGemini_(TEXT_MODEL, { contents: [{ parts: [{ text: prompt }] }], generationConfig: { responseMimeType: 'application/json' } }); const text = data.candidates[0].content.parts[0].text; return JSON.parse(text); } // ---- Image: returns a data: URL (base64) ---- function generateImage_(scene, characters) { const prompt = 'Comic book panel illustration, bold black outlines, vibrant colors, consistent art style, ' + 'no text, no speech bubbles, no captions. ' + 'Characters: ' + characters + '. Scene: ' + scene; const data = callGemini_(IMAGE_MODEL, { contents: [{ parts: [{ text: prompt }] }] }); const parts = (data.candidates && data.candidates[0] && data.candidates[0].content.parts) || []; const img = parts.find(function (p) { return p.inlineData; }); if (!img) return { image: null, error: JSON.stringify(data).slice(0, 300) }; return { image: 'data:' + img.inlineData.mimeType + ';base64,' + img.inlineData.data }; } function doGet(e) { const p = e.parameter || {}; let out; try { if (p.mode === 'image') { out = generateImage_(p.scene || '', p.characters || ''); } else { out = generateStory_(p.idea || 'four friends find a mysterious book'); } } catch (err) { out = { error: String(err) }; } return ContentService.createTextOutput(JSON.stringify(out)) .setMimeType(ContentService.MimeType.JSON); }
